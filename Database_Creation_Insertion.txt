-- ============================================================
-- 1. CREATE DATABASE
-- ============================================================

DROP DATABASE IF EXISTS chefconnect_db;

CREATE DATABASE chefconnect_db;

USE chefconnect_db;


-- ============================================================
-- 2. CREATE LOCATIONS TABLE
-- ============================================================

CREATE TABLE locations (
    location_id INT PRIMARY KEY AUTO_INCREMENT,
    area VARCHAR(100) NOT NULL,
    city VARCHAR(100) NOT NULL
);


-- ============================================================
-- 3. CREATE USERS TABLE
-- ============================================================

CREATE TABLE users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,
    email VARCHAR(100) NOT NULL UNIQUE,
    role ENUM('admin', 'customer', 'chef') NOT NULL
);


-- ============================================================
-- 4. CREATE CUSTOMERS TABLE
-- ============================================================

CREATE TABLE customers (
    customer_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL UNIQUE,
    location_id INT NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    phone VARCHAR(15),

    CONSTRAINT fk_customer_user
        FOREIGN KEY (user_id)
        REFERENCES users(user_id),

    CONSTRAINT fk_customer_location
        FOREIGN KEY (location_id)
        REFERENCES locations(location_id)
);


-- ============================================================
-- 5. CREATE CHEFS TABLE
-- ============================================================

CREATE TABLE chefs (
    chef_id INT PRIMARY KEY AUTO_INCREMENT,
    chef_name VARCHAR(100) NOT NULL,
    user_id INT NOT NULL UNIQUE,
    location_id INT NOT NULL,
    experience_years INT NOT NULL,
    verification_status ENUM('Pending', 'Verified', 'Rejected') NOT NULL,

    CONSTRAINT fk_chef_user
        FOREIGN KEY (user_id)
        REFERENCES users(user_id),

    CONSTRAINT fk_chef_location
        FOREIGN KEY (location_id)
        REFERENCES locations(location_id)
);


-- ============================================================
-- 6. CREATE CUISINES TABLE
-- ============================================================

CREATE TABLE cuisines (
    cuisine_id INT PRIMARY KEY AUTO_INCREMENT,
    cuisine_name VARCHAR(100) NOT NULL UNIQUE
);


-- ============================================================
-- 7. CREATE CHEF_SPECIALITIES TABLE  (composite primary key)
-- ============================================================

CREATE TABLE chef_specialities (
    chef_id INT NOT NULL,
    cuisine_id INT NOT NULL,
    skill_level ENUM('Beginner', 'Intermediate', 'Expert') NOT NULL,

    PRIMARY KEY (chef_id, cuisine_id),

    CONSTRAINT fk_speciality_chef
        FOREIGN KEY (chef_id)
        REFERENCES chefs(chef_id),

    CONSTRAINT fk_speciality_cuisine
        FOREIGN KEY (cuisine_id)
        REFERENCES cuisines(cuisine_id)
);


-- ============================================================
-- 8. CREATE CHEF_AVAILABILITY TABLE
-- ============================================================

CREATE TABLE chef_availability (
    chef_id INT NOT NULL,
    availability_dt DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    status ENUM('Available', 'Booked', 'Unavailable') NOT NULL,

    PRIMARY KEY (chef_id, availability_dt, start_time),

    CONSTRAINT fk_availability_chef
        FOREIGN KEY (chef_id)
        REFERENCES chefs(chef_id)
);


-- ============================================================
-- 9. CREATE MENUS TABLE
-- ============================================================

CREATE TABLE menus (
    menu_id INT PRIMARY KEY AUTO_INCREMENT,
    chef_id INT NOT NULL,
    menu_name VARCHAR(100) NOT NULL,
    price_per_person DECIMAL(10,2) NOT NULL,
    diet_flag ENUM('Veg', 'Non-Veg', 'Vegan') NOT NULL,

    CONSTRAINT uq_menu_chef UNIQUE (menu_id, chef_id),

    CONSTRAINT fk_menu_chef
        FOREIGN KEY (chef_id)
        REFERENCES chefs(chef_id)
);


-- ============================================================
-- 10. CREATE EVENTS TABLE
-- ============================================================

CREATE TABLE events (
    event_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    event_date DATE NOT NULL,
    service_time TIME NOT NULL,
    event_type VARCHAR(50) NOT NULL,
    service_type ENUM('Event', 'Personal Chef') NOT NULL,
    guest_count INT NOT NULL,
    requirements TEXT,

    CONSTRAINT fk_event_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
);


-- ============================================================
-- 11. CREATE BOOKINGS TABLE
-- ============================================================

CREATE TABLE bookings (
    booking_id INT PRIMARY KEY AUTO_INCREMENT,
    event_id INT NOT NULL,
    chef_id INT NOT NULL,
    menu_id INT NOT NULL,
    booking_status ENUM('Pending', 'Confirmed', 'Completed', 'Cancelled') NOT NULL,
    ingredient_source ENUM('Customer', 'Chef') NOT NULL DEFAULT 'Customer',
    ingredient_cost DECIMAL(10,2) NOT NULL DEFAULT 0,

    CONSTRAINT chk_ingredient
        CHECK (ingredient_source = 'Chef' OR ingredient_cost = 0),

    CONSTRAINT fk_booking_event
        FOREIGN KEY (event_id)
        REFERENCES events(event_id),

    CONSTRAINT fk_booking_chef
        FOREIGN KEY (chef_id)
        REFERENCES chefs(chef_id),

    CONSTRAINT fk_booking_menu_chef
        FOREIGN KEY (menu_id, chef_id)
        REFERENCES menus(menu_id, chef_id)
);


-- ============================================================
-- 12. CREATE PAYMENTS TABLE
-- ============================================================

CREATE TABLE payments (
    booking_id INT NOT NULL,
    payment_method ENUM('Cash', 'UPI', 'Card', 'Net Banking') NOT NULL,
    payment_status ENUM('Success', 'Pending', 'Failed') NOT NULL,
    payment_type ENUM('Service', 'Ingredients') NOT NULL DEFAULT 'Service',
    amount DECIMAL(10,2) NOT NULL,
    payment_date DATETIME NOT NULL,

    PRIMARY KEY (booking_id, payment_date, payment_type),

    CONSTRAINT fk_payment_booking
        FOREIGN KEY (booking_id)
        REFERENCES bookings(booking_id)
);


-- ============================================================
-- 13. CREATE REVIEWS TABLE
-- ============================================================

CREATE TABLE reviews (
    booking_id INT PRIMARY KEY,
    rating INT NOT NULL,

    CONSTRAINT chk_review_rating CHECK (rating BETWEEN 1 AND 5),

    CONSTRAINT fk_review_booking
        FOREIGN KEY (booking_id)
        REFERENCES bookings(booking_id)
);

-- 26. VIEWS FOR THE DERIVED ATTRIBUTES (calculated, never stored)
-- ============================================================

-- BOOKINGS.booking_total, grand_total, balance_due, chef_payout, platform_commission
CREATE VIEW booking_financials AS
SELECT b.booking_id,
       m.price_per_person * e.guest_count AS booking_total,
       m.price_per_person * e.guest_count + b.ingredient_cost AS grand_total,
       m.price_per_person * e.guest_count
         - IFNULL((SELECT SUM(p.amount) FROM payments p
                   WHERE p.booking_id = b.booking_id
                     AND p.payment_status = 'Success'
                     AND p.payment_type = 'Service'), 0) AS balance_due,
       ROUND(m.price_per_person * e.guest_count * 0.85 + b.ingredient_cost, 2) AS chef_payout,
       ROUND(m.price_per_person * e.guest_count * 0.15, 2) AS platform_commission
FROM bookings b
JOIN menus  m ON b.menu_id  = m.menu_id
JOIN events e ON b.event_id = e.event_id;

-- CHEFS.avg_rating
CREATE VIEW chef_avg_rating AS
SELECT ch.chef_id, ch.chef_name, ROUND(AVG(r.rating), 2) AS avg_rating
FROM chefs ch
LEFT JOIN bookings b ON b.chef_id = ch.chef_id
LEFT JOIN reviews  r ON r.booking_id = b.booking_id
GROUP BY ch.chef_id, ch.chef_name;
