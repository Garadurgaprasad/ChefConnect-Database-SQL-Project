-- ============================================================
-- A. SUBQUERIES
-- Q71 - Q80
-- A subquery is a SELECT written inside another query.

-- Q71. Find menus priced above the average menu price.
SELECT menu_name, price_per_person
FROM menus
WHERE price_per_person > (SELECT AVG(price_per_person) FROM menus);

-- Q72. Find the chef with the most experience.
SELECT chef_name, experience_years
FROM chefs
WHERE experience_years = (SELECT MAX(experience_years) FROM chefs);

-- Q73. Find events that have at least one booking (IN).
SELECT event_id, event_type, event_date
FROM events
WHERE event_id IN (SELECT event_id FROM bookings);

-- Q74. Find customers who have planned at least one event (EXISTS).
SELECT customer_id, first_name
FROM customers c
WHERE EXISTS (
    SELECT 1
    FROM events e
    WHERE e.customer_id = c.customer_id
);

-- Q75. Find bookings whose event has more guests than the average guest count.
SELECT b.booking_id, e.event_type, e.guest_count
FROM bookings b
JOIN events e ON b.event_id = e.event_id
WHERE e.guest_count > (SELECT AVG(guest_count) FROM events)
ORDER BY b.booking_id;

-- Q76. Find chefs who offer a Vegan menu.
SELECT chef_name
FROM chefs
WHERE chef_id IN (
    SELECT chef_id
    FROM menus
    WHERE diet_flag = 'Vegan'
);

-- Q77. Find chefs whose average rating is higher than the overall average rating.
SELECT ch.chef_name, ROUND(AVG(r.rating), 2) AS avg_rating
FROM reviews r
JOIN bookings b ON r.booking_id = b.booking_id
JOIN chefs ch ON b.chef_id = ch.chef_id
GROUP BY ch.chef_id, ch.chef_name
HAVING AVG(r.rating) > (SELECT AVG(rating) FROM reviews)
ORDER BY ch.chef_id;

-- Q78. Find customers who have never planned an event (NOT IN).
SELECT customer_id, first_name
FROM customers
WHERE customer_id NOT IN (SELECT customer_id FROM events);

-- Q79. Find events that do not have any successful service payment (NOT EXISTS).
SELECT e.event_id, e.event_type, e.event_date
FROM events e
WHERE NOT EXISTS (
    SELECT 1
    FROM bookings b
    JOIN payments p ON b.booking_id = p.booking_id
    WHERE b.event_id = e.event_id
      AND p.payment_status = 'Success' AND p.payment_type = 'Service'
)
ORDER BY e.event_id;

-- Q80. Show every chef with the number of bookings (subquery in the SELECT list).
SELECT ch.chef_name,
       (SELECT COUNT(*) FROM bookings b WHERE b.chef_id = ch.chef_id) AS total_bookings
FROM chefs ch
ORDER BY total_bookings DESC, ch.chef_name;


-- ============================================================
-- B. VIEWS
-- Q81 - Q84
-- A view is a saved SELECT. It stores no data, so it always
-- shows the current rows of the tables underneath it.
-- (booking_financials and chef_avg_rating already exist from
--  Database_Creation_Insertion.txt and are reused below.)
-- ============================================================

-- Q81. Create a view that joins everything about a booking in one place
--      (customer, chef, menu, event and money columns).
CREATE OR REPLACE VIEW v_booking_details AS
SELECT b.booking_id,
       e.event_id,
       c.customer_id,
       c.first_name       AS customer_name,
       cl.city            AS customer_city,
       ch.chef_id,
       ch.chef_name,
       m.menu_name,
       m.diet_flag,
       e.event_type,
       e.service_type,
       e.event_date,
       e.service_time,
       e.guest_count,
       b.booking_status,
       b.ingredient_source,
       bf.booking_total,
       bf.grand_total,
       bf.balance_due
FROM bookings b
JOIN events    e  ON b.event_id    = e.event_id
JOIN customers c  ON e.customer_id = c.customer_id
JOIN locations cl ON c.location_id = cl.location_id
JOIN chefs     ch ON b.chef_id     = ch.chef_id
JOIN menus     m  ON b.menu_id     = m.menu_id
JOIN booking_financials bf ON b.booking_id = bf.booking_id;

SELECT * FROM v_booking_details ORDER BY event_date;


-- Q82. Using the view: show the outstanding balance of every non-cancelled booking.
SELECT booking_id, customer_name, booking_status,
       booking_total,
       booking_total - balance_due AS paid_amount,
       balance_due
FROM v_booking_details
WHERE booking_status <> 'Cancelled'
ORDER BY balance_due DESC, booking_id;


-- Q83. Create a chef scorecard view (completed bookings, average rating, total payout).
CREATE OR REPLACE VIEW v_chef_scorecard AS
SELECT ch.chef_id,
       ch.chef_name,
       COUNT(b.booking_id) AS completed_bookings,
       car.avg_rating,
       ROUND(IFNULL(SUM(bf.chef_payout), 0), 2) AS total_payout
FROM chefs ch
LEFT JOIN bookings b            ON ch.chef_id = b.chef_id AND b.booking_status = 'Completed'
LEFT JOIN booking_financials bf ON b.booking_id = bf.booking_id
LEFT JOIN chef_avg_rating car   ON ch.chef_id = car.chef_id
GROUP BY ch.chef_id, ch.chef_name, car.avg_rating;

SELECT * FROM v_chef_scorecard ORDER BY total_payout DESC;


-- Q84. Create a monthly revenue view (successful service payments only).
CREATE OR REPLACE VIEW v_monthly_revenue AS
SELECT DATE_FORMAT(payment_date, '%Y-%m') AS month,
       COUNT(*)    AS payments_received,
       SUM(amount) AS revenue
FROM payments
WHERE payment_status = 'Success' AND payment_type = 'Service'
GROUP BY DATE_FORMAT(payment_date, '%Y-%m');

SELECT * FROM v_monthly_revenue ORDER BY month;

-- Q85. Procedure: list all bookings of one chef.
DROP PROCEDURE IF EXISTS sp_chef_bookings;

DELIMITER $$
CREATE PROCEDURE sp_chef_bookings(IN p_chef_id INT)
BEGIN
    SELECT booking_id, customer_name, event_type, service_type,
           event_date, booking_status, booking_total, balance_due
    FROM v_booking_details
    WHERE chef_id = p_chef_id
    ORDER BY event_date;
END$$
DELIMITER ;

CALL sp_chef_bookings(3);


-- Q86. Procedure: customer statement - every booking of one customer with the money paid and due.
DROP PROCEDURE IF EXISTS sp_customer_statement;

DELIMITER $$
CREATE PROCEDURE sp_customer_statement(IN p_customer_id INT)
BEGIN
    SELECT booking_id, chef_name, menu_name, event_type, event_date, booking_status,
           booking_total,
           booking_total - balance_due AS paid_amount,
           balance_due
    FROM v_booking_details
    WHERE customer_id = p_customer_id
    ORDER BY event_date;
END$$
DELIMITER ;

CALL sp_customer_statement(2);


-- Q87. Procedure: monthly revenue of a given year, with the share of the year's revenue.
DROP PROCEDURE IF EXISTS sp_monthly_revenue;

DELIMITER $$
CREATE PROCEDURE sp_monthly_revenue(IN p_year INT)
BEGIN
    SELECT month, revenue,
           ROUND(revenue * 100.0 / (SELECT SUM(revenue)
                                    FROM v_monthly_revenue
                                    WHERE month LIKE CONCAT(p_year, '-%')), 2) AS share_of_year_pct
    FROM v_monthly_revenue
    WHERE month LIKE CONCAT(p_year, '-%')
    ORDER BY month;
END$$
DELIMITER ;

CALL sp_monthly_revenue(2026);


-- Q88. Procedure: find verified chefs of a city who are free on a date
--      (an 'Available' slot and no active booking on that date).
DROP PROCEDURE IF EXISTS sp_available_chefs;

DELIMITER $$
CREATE PROCEDURE sp_available_chefs(IN p_city VARCHAR(100), IN p_date DATE)
BEGIN
    SELECT ch.chef_id, ch.chef_name, ch.experience_years,
           a.start_time, a.end_time
    FROM chefs ch
    JOIN locations l         ON ch.location_id = l.location_id
    JOIN chef_availability a ON ch.chef_id = a.chef_id
    WHERE l.city = p_city
      AND ch.verification_status = 'Verified'
      AND a.availability_dt = p_date
      AND a.status = 'Available'
      AND NOT EXISTS (
            SELECT 1
            FROM bookings b
            JOIN events e ON b.event_id = e.event_id
            WHERE b.chef_id = ch.chef_id
              AND e.event_date = p_date
              AND b.booking_status <> 'Cancelled');
END$$
DELIMITER ;

CALL sp_available_chefs('Bengaluru', '2026-10-20');

-- Q90. Procedure: record a payment and return the balance still due.
--      (The payment rules are checked by the trigger in Q96.)
DROP PROCEDURE IF EXISTS sp_record_payment;

DELIMITER $$
CREATE PROCEDURE sp_record_payment(
    IN  p_booking_id  INT,
    IN  p_method      VARCHAR(20),
    IN  p_type        VARCHAR(20),
    IN  p_amount      DECIMAL(10,2),
    OUT p_balance_due DECIMAL(10,2))
BEGIN
    INSERT INTO payments (booking_id, payment_method, payment_status, payment_type, amount, payment_date)
    VALUES (p_booking_id, p_method, 'Success', p_type, p_amount, NOW());

    SELECT balance_due INTO p_balance_due
    FROM booking_financials
    WHERE booking_id = p_booking_id;
END$$
DELIMITER ;

SELECT @balance_left AS balance_due_after_payment;
ROLLBACK;


-- Q91. Procedure: cancel a booking and return the amount that was paid for it.
--      (Simple rule for this project: the paid service amount is the refund amount.
--       The trigger in Q93 stops Completed or Cancelled bookings from being changed.)
DROP PROCEDURE IF EXISTS sp_cancel_booking;

DELIMITER $$
CREATE PROCEDURE sp_cancel_booking(
    IN  p_booking_id INT,
    OUT p_refund     DECIMAL(10,2))
BEGIN
    SELECT IFNULL(SUM(amount), 0) INTO p_refund
    FROM payments
    WHERE booking_id = p_booking_id
      AND payment_status = 'Success' AND payment_type = 'Service';

    UPDATE bookings SET booking_status = 'Cancelled' WHERE booking_id = p_booking_id;
END$$
DELIMITER ;

-- Test (rolled back): cancel booking 6 (Confirmed, 5625 already paid).
START TRANSACTION;
CALL sp_cancel_booking(6, @refund);
SELECT @refund AS refund_amount;
SELECT booking_id, booking_status FROM bookings WHERE booking_id = 6;
ROLLBACK;


-- ============================================================
-- D. TRIGGERS
-- Q92 - Q97
-- A trigger runs automatically BEFORE or AFTER an INSERT / UPDATE / DELETE.
-- NEW = the new row values, OLD = the row values before the change.
-- Tests that change data are wrapped in START TRANSACTION ... ROLLBACK.
-- Tests that are meant to FAIL are written as comments with the expected error.
-- ============================================================

-- Q92. Trigger (AFTER UPDATE): keep an audit log of every booking status change.
--      First create the small log table the trigger writes into.
CREATE TABLE IF NOT EXISTS booking_status_log (
    log_id      INT PRIMARY KEY AUTO_INCREMENT,
    booking_id  INT NOT NULL,
    old_status  VARCHAR(20) NOT NULL,
    new_status  VARCHAR(20) NOT NULL,
    changed_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_log_booking
        FOREIGN KEY (booking_id)
        REFERENCES bookings(booking_id)
);

DROP TRIGGER IF EXISTS trg_booking_status_log;

DELIMITER $$
CREATE TRIGGER trg_booking_status_log
AFTER UPDATE ON bookings
FOR EACH ROW
BEGIN
    IF OLD.booking_status <> NEW.booking_status THEN
        INSERT INTO booking_status_log (booking_id, old_status, new_status)
        VALUES (OLD.booking_id, OLD.booking_status, NEW.booking_status);
    END IF;
END$$
DELIMITER ;

-- Test: confirm the pending booking 8 and read the log.
START TRANSACTION;
UPDATE bookings SET booking_status = 'Confirmed' WHERE booking_id = 8;
SELECT * FROM booking_status_log;
ROLLBACK;


-- Q93. Trigger (BEFORE UPDATE): protect the booking life cycle.
--      Completed and Cancelled bookings are final, and a booking
--      must be Confirmed before it can be Completed.
DROP TRIGGER IF EXISTS trg_booking_status_flow;

DELIMITER $$
CREATE TRIGGER trg_booking_status_flow
BEFORE UPDATE ON bookings
FOR EACH ROW
BEGIN
    IF OLD.booking_status <> NEW.booking_status THEN
        IF OLD.booking_status IN ('Completed', 'Cancelled') THEN
            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Completed or Cancelled bookings cannot change status';
        END IF;

        IF OLD.booking_status = 'Pending' AND NEW.booking_status = 'Completed' THEN
            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'A booking must be Confirmed before it is Completed';
        END IF;
    END IF;
END$$
DELIMITER ;

-- Q95. Trigger (AFTER INSERT): when a booking is created, mark the chef's
--      'Available' slot on the event date as 'Booked'.
DROP TRIGGER IF EXISTS trg_booking_block_slot;

DELIMITER $$
CREATE TRIGGER trg_booking_block_slot
AFTER INSERT ON bookings
FOR EACH ROW
BEGIN
    UPDATE chef_availability
    SET status = 'Booked'
    WHERE chef_id = NEW.chef_id
      AND availability_dt = (SELECT event_date FROM events WHERE event_id = NEW.event_id)
      AND status = 'Available';
END$$

-- Q96. Trigger (BEFORE INSERT): check every new payment.
--      The amount must be positive, a cancelled booking takes no payment, and a
--      successful Service payment cannot be more than the balance due.
DROP TRIGGER IF EXISTS trg_payment_check;

DELIMITER $$
CREATE TRIGGER trg_payment_check
BEFORE INSERT ON payments
FOR EACH ROW
BEGIN
    DECLARE v_status  VARCHAR(20);
    DECLARE v_balance DECIMAL(10,2);

    IF NEW.amount <= 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Payment amount must be greater than zero';
    END IF;

    SELECT b.booking_status, bf.balance_due
      INTO v_status, v_balance
    FROM bookings b
    JOIN booking_financials bf ON b.booking_id = bf.booking_id
    WHERE b.booking_id = NEW.booking_id;

    IF v_status = 'Cancelled' THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Cannot pay for a cancelled booking';
    END IF;

    IF NEW.payment_type = 'Service' AND NEW.payment_status = 'Success'
       AND NEW.amount > v_balance THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Payment is more than the balance due';
    END IF;
END$$
DELIMITER ;

-- Q98. CASE: segment customers by total successful spend (Premium / Regular / Basic).
SELECT c.first_name,
       IFNULL(SUM(p.amount), 0) AS total_spend,
       CASE
           WHEN IFNULL(SUM(p.amount), 0) >= 30000 THEN 'Premium'
           WHEN IFNULL(SUM(p.amount), 0) >= 15000 THEN 'Regular'
           ELSE 'Basic'
       END AS segment
FROM customers c
LEFT JOIN events e   ON c.customer_id = e.customer_id
LEFT JOIN bookings b ON e.event_id = b.event_id
LEFT JOIN payments p ON b.booking_id = p.booking_id
                    AND p.payment_status = 'Success' AND p.payment_type = 'Service'
GROUP BY c.customer_id, c.first_name
ORDER BY total_spend DESC, c.first_name;


-- Q99. Stored FUNCTION: return the booking total (menu price x guests) of a booking.
--      A function returns one value, so it can be used inside SELECT and WHERE.
DROP FUNCTION IF EXISTS fn_booking_total;

DELIMITER $$
CREATE FUNCTION fn_booking_total(p_booking_id INT)
RETURNS DECIMAL(10,2)
READS SQL DATA
BEGIN
    DECLARE v_total DECIMAL(10,2);

    SELECT m.price_per_person * e.guest_count
      INTO v_total
    FROM bookings b
    JOIN menus  m ON b.menu_id  = m.menu_id
    JOIN events e ON b.event_id = e.event_id
    WHERE b.booking_id = p_booking_id;

    RETURN v_total;
END$$
DELIMITER ;

-- Use the function: bookings worth more than 10000.
SELECT booking_id, booking_status, fn_booking_total(booking_id) AS booking_total
FROM bookings
WHERE fn_booking_total(booking_id) > 10000
ORDER BY booking_total DESC;
