-- ============================================================
-- MAIN TABLE OF THE PROJECT : BOOKINGS
-- DATABASE : chefconnect_db
-- ============================================================
--
-- WHY BOOKINGS IS THE MAIN (PRIMARY) TABLE
-- ----------------------------------------
-- ChefConnect exists to turn "a customer's event" into "a chef cooking a
-- menu at that event". BOOKINGS is the only table that holds all three:
--
--     EVENTS  (what the customer wants, and who the customer is)
--     CHEFS   (who will cook)
--     MENUS   (what will be cooked, and at what price)
--
-- and everything that happens afterwards hangs off it:
--
--     PAYMENTS  (money paid for the booking)
--     REVIEWS   (rating given after the booking)
--
-- So ONE booking_id is enough to reach every other table in the project.
--
--
-- THE TWO HIRING OPTIONS (chosen when the customer books)
-- -------------------------------------------------------
--  1. SERVICE TYPE  (events.service_type, with events.event_date + events.service_time)
--       'Event'          chef cooks for an occasion (party, wedding, corporate lunch)
--       'Personal Chef'  chef comes to the customer's HOME at a specific date and time
--                        and cooks a fresh, healthy meal - no delivery app, no restaurant
--  2. INGREDIENT SOURCE  (bookings.ingredient_source and bookings.ingredient_cost)
--       'Customer'  chef cooks with the ingredients already present at the house (cost = 0)
--       'Chef'      chef fetches the ingredients; the customer pays the bill as a separate
--                   payments row with payment_type = 'Ingredients' and the chef is
--                   reimbursed in full (it is never counted as platform revenue)
--
--
-- RELATION MAP  (arrows point from the child table to the table it references)
-- ---------------------------------------------------------------------------
--
--   USERS ◄── CUSTOMERS ◄── EVENTS ◄──┐
--                │                    │
--   LOCATIONS ◄──┤              ┌─────┴─────┐
--                │              │ BOOKINGS  │ ◄── PAYMENTS  (1 booking : many payments)
--   LOCATIONS ◄── CHEFS ◄───────┤  (MAIN)   │ ◄── REVIEWS   (1 booking : 0..1 review)
--                │              └─────┬─────┘
--   USERS ◄──────┘                    │
--                                     └──► MENUS (menu must belong to the same chef)
--   CHEFS ◄── CHEF_AVAILABILITY
--   CHEFS ◄── CHEF_SPECIALITIES ──► CUISINES
--
--
-- HOW EACH TABLE IS LINKED TO BOOKINGS
-- ------------------------------------
--  Table              Link to BOOKINGS                    Information fetched
--  -----------------  ----------------------------------  ---------------------------------
--  EVENTS             bookings.event_id = events.event_id event type, date, guests, requirements
--  CUSTOMERS          via EVENTS (events.customer_id)     who booked: name, phone
--  LOCATIONS          via CUSTOMERS / CHEFS (location_id) customer city, chef city
--  CHEFS              bookings.chef_id = chefs.chef_id    chef name, experience, verification
--  MENUS              bookings.menu_id = menus.menu_id    menu name, diet type, price per person
--  PAYMENTS           payments.booking_id                 amount, method, status, date
--  REVIEWS            reviews.booking_id                  rating
--  CHEF_AVAILABILITY  via CHEFS (chef_id)                 dates the booked chef is free
--  CHEF_SPECIALITIES  via CHEFS (chef_id) -> CUISINES     cuisines the booked chef cooks
--  USERS              via CUSTOMERS / CHEFS (user_id)     login email and role
--
-- Integrity rule: FOREIGN KEY (menu_id, chef_id) REFERENCES menus(menu_id, chef_id)
-- guarantees that the menu in a booking is always offered by the booked chef.
--
-- Derived values (calculated, never stored):
--   booking_total = menus.price_per_person * events.guest_count      (the service charge)
--   grand_total   = booking_total + bookings.ingredient_cost
--   balance_due   = booking_total - SUM(successful 'Service' payments)
--   chef_payout   = booking_total * 0.85  (+ ingredient_cost, reimbursed in full)
--   platform_commission = booking_total * 0.15
--
-- Change the values below to look up any other booking / customer / chef.

USE chefconnect_db;

SET @bid  = 3;   -- booking to inspect
SET @cid  = 1;   -- customer to inspect
SET @chid = 3;   -- chef to inspect
SET @pbid = 12;  -- personal-chef booking to inspect (chef bought the ingredients)
SET @cid2 = 2;   -- customer who wants to hire a personal chef


-- M1. Everything about ONE booking in a single row (events + customer + location + chef + menu).
SELECT b.booking_id, b.booking_status,
       e.event_type, e.event_date, e.guest_count,
       c.first_name AS customer, c.phone, l.city AS customer_city,
       ch.chef_name, m.menu_name, m.diet_flag, m.price_per_person,
       m.price_per_person * e.guest_count AS booking_total,
       e.service_type, e.service_time, b.ingredient_source, b.ingredient_cost
FROM bookings b
JOIN events e ON b.event_id = e.event_id
JOIN customers c ON e.customer_id = c.customer_id
JOIN locations l ON c.location_id = l.location_id
JOIN chefs ch ON b.chef_id = ch.chef_id
JOIN menus m ON b.menu_id = m.menu_id
WHERE b.booking_id = @bid;
--
-- RESULT (@bid = 3, @cid = 1, @chid = 3, @pbid = 12, @cid2 = 2):
-- +------------+----------------+-----------------+------------+-------------+----------+------------+---------------+----------------+-------------------------+-----------+------------------+---------------+--------------+--------------+-------------------+-----------------+
-- | booking_id | booking_status | event_type      | event_date | guest_count | customer | phone      | customer_city | chef_name      | menu_name               | diet_flag | price_per_person | booking_total | service_type | service_time | ingredient_source | ingredient_cost |
-- +------------+----------------+-----------------+------------+-------------+----------+------------+---------------+----------------+-------------------------+-----------+------------------+---------------+--------------+--------------+-------------------+-----------------+
-- |          3 | Completed      | Corporate Lunch | 2026-09-05 |          50 | Arjun    | 9876500003 | Visakhapatnam | Harpreet Singh | North Indian Veg Deluxe | Veg       |           500.00 |      25000.00 | Event        | 13:00:00     | Chef              |         6000.00 |
-- +------------+----------------+-----------------+------------+-------------+----------+------------+---------------+----------------+-------------------------+-----------+------------------+---------------+--------------+--------------+-------------------+-----------------+
-- 1 row in set

-- M3. The rating given for that booking (BOOKINGS -> REVIEWS).
SELECT booking_id, rating

-- | booking_id | rating |

-- M4. Money summary of that booking: total, paid, balance, chef payout and commission.
SELECT b.booking_id,
       m.price_per_person * e.guest_count AS booking_total,
       IFNULL((SELECT SUM(p.amount) FROM payments p
               WHERE p.booking_id = b.booking_id AND p.payment_status = 'Success' AND p.payment_type = 'Service'), 0) AS paid_amount,
       m.price_per_person * e.guest_count
         - IFNULL((SELECT SUM(p.amount) FROM payments p
                   WHERE p.booking_id = b.booking_id AND p.payment_status = 'Success' AND p.payment_type = 'Service'), 0) AS balance_due,
       ROUND(m.price_per_person * e.guest_count * 0.85, 2) AS chef_payout,
       ROUND(m.price_per_person * e.guest_count * 0.15, 2) AS platform_commission
FROM bookings b
JOIN menus m ON b.menu_id = m.menu_id
JOIN events e ON b.event_id = e.event_id
WHERE b.booking_id = @bid;
--
-- RESULT (@bid = 3, @cid = 1, @chid = 3, @pbid = 12, @cid2 = 2):
-- +------------+---------------+-------------+-------------+-------------+---------------------+
-- | booking_id | booking_total | paid_amount | balance_due | chef_payout | platform_commission |
-- +------------+---------------+-------------+-------------+-------------+---------------------+
-- |          3 |      25000.00 |    25000.00 |        0.00 |    21250.00 |             3750.00 |
-- +------------+---------------+-------------+-------------+-------------+---------------------+
-- 1 row in set


-- M5. Customer side: every booking made by one customer (CUSTOMERS -> EVENTS -> BOOKINGS).

JOIN menus m ON b.menu_id = m.menu_id
WHERE c.customer_id = @cid

-- M11. Compare the two service types: bookings, guests and service value (cancelled excluded).
SELECT e.service_type,
       COUNT(*) AS total_bookings,
       SUM(e.guest_count) AS total_guests,
       SUM(m.price_per_person * e.guest_count) AS service_value
FROM bookings b
JOIN events e ON b.event_id = e.event_id
JOIN menus m ON b.menu_id = m.menu_id
WHERE b.booking_status <> 'Cancelled'
GROUP BY e.service_type;

-- M13. Bookings where the chef fetches the ingredients: cost recorded and amount paid back so far.

-- M14. Ingredient source by service type: how many bookings use home ingredients and how many chef-bought.
SELECT e.service_type, b.ingredient_source,
       COUNT(*) AS total_bookings,

-- M15. Full bill of one booking: service charge + ingredients, what is paid, and what the chef receives.
SELECT b.booking_id, e.service_type, b.ingredient_source,
       m.price_per_person * e.guest_count AS service_charge,
       b.ingredient_cost,
       m.price_per_person * e.guest_count + b.ingredient_cost AS grand_total,
       IFNULL((SELECT SUM(p.amount) FROM payments p
               WHERE p.booking_id = b.booking_id AND p.payment_status = 'Success'), 0) AS total_paid,
       ROUND(m.price_per_person * e.guest_count * 0.85 + b.ingredient_cost, 2) AS chef_receives,
       ROUND(m.price_per_person * e.guest_count * 0.15, 2) AS platform_commission
FROM bookings b
JOIN events e ON b.event_id = e.event_id
JOIN menus m ON b.menu_id = m.menu_id
WHERE b.booking_id = @pbid;
--
-- RESULT (@bid = 3, @cid = 1, @chid = 3, @pbid = 12, @cid2 = 2):
-- +------------+---------------+-------------------+----------------+-----------------+-------------+------------+---------------+---------------------+
-- | booking_id | service_type  | ingredient_source | service_charge | ingredient_cost | grand_total | total_paid | chef_receives | platform_commission |
-- +------------+---------------+-------------------+----------------+-----------------+-------------+------------+---------------+---------------------+
-- |         12 | Personal Chef | Chef              |        1400.00 |         1200.00 |     2600.00 |    2600.00 |       2390.00 |              210.00 |
-- +------------+---------------+-------------------+----------------+-----------------+-------------+------------+---------------+---------------------+
-- 1 row in set


-- M16. Hiring a personal chef: verified chefs in the customer's own city free on a given date.
SELECT c.first_name AS customer, l.city, ch.chef_name, ch.experience_years,
       a.availability_dt, a.start_time, a.end_time
FROM customers c
JOIN locations l ON c.location_id = l.location_id
JOIN chefs ch ON ch.location_id = l.location_id
JOIN chef_availability a ON a.chef_id = ch.chef_id
WHERE c.customer_id = @cid2
  AND ch.verification_status = 'Verified'
  AND a.status = 'Available'
  AND a.availability_dt = '2026-10-15';
--
-- RESULT (@bid = 3, @cid = 1, @chid = 3, @pbid = 12, @cid2 = 2):
-- +----------+-----------+----------------+------------------+-----------------+------------+----------+

-- M17. Menus of that chef with the cost for a 2-person home meal.
SELECT ch.chef_name, m.menu_name, m.diet_flag, m.price_per_person,
       m.price_per_person * 2 AS cost_for_2_people
FROM chefs ch
JOIN menus m ON ch.chef_id = m.chef_id
WHERE ch.chef_id = @chid
ORDER BY m.price_per_person;

-- M18. Monthly money split: service revenue versus ingredient reimbursements paid through the platform.
SELECT DATE_FORMAT(payment_date, '%Y-%m') AS month,
       SUM(CASE WHEN payment_type = 'Service' THEN amount ELSE 0 END) AS service_revenue,
       SUM(CASE WHEN payment_type = 'Ingredients' THEN amount ELSE 0 END) AS ingredient_reimbursement
FROM payments
WHERE payment_status = 'Success'
GROUP BY DATE_FORMAT(payment_date, '%Y-%m')
ORDER BY month;
